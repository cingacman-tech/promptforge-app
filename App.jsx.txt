export default function App() {
  const [activeTab, setActiveTab] = useState('studio'); // 'studio', 'templates', 'monetization', 'analytics'
  
  // Custom User API Key State (Optional overrides)
  const [userApiKey, setUserApiKey] = useState('');
  const [showKeyModal, setShowKeyModal] = useState(false);
  const [showExportModal, setShowExportModal] = useState(false);

  // Studio Builder State
  const [promptTitle, setPromptTitle] = useState('SEO & Social Viral Engine');
  const [selectedModel, setSelectedModel] = useState('gemini-3-flash-preview');
  const [promptTemplate, setPromptTemplate] = useState(
    'Act as an expert content marketer for {{industry}}. Write a high-converting {{content_type}} tailored to target audience: {{target_audience}}. Focus on tone: {{tone}}.'
  );

  // Variable Metadata map (label, default value)
  const [variableMeta, setVariableMeta] = useState({
    industry: { label: 'Industry Sector', default: 'SaaS Fintech' },
    content_type: { label: 'Content Format', default: 'LinkedIn Post' },
    target_audience: { label: 'Target Audience', default: 'Startup Founders' },
    tone: { label: 'Voice & Style', default: 'Authoritative & Actionable' }
  });

  // User input values for test execution
  const [inputValues, setInputValues] = useState({});

  // Auto-extract dynamic variables from {{variable_name}} syntax in prompt template
  const extractedVariables = useMemo(() => {
    const matches = promptTemplate.match(/\{\{([a-zA-Z0-9_]+)\}\}/g) || [];
    const keys = Array.from(new Set(matches.map(m => m.replace(/[\{\}]/g, '').trim()))).filter(Boolean);
    return keys;
  }, [promptTemplate]);

  // Ensure variable metadata and input values stay synced when template changes
  useEffect(() => {
    setVariableMeta(prevMeta => {
      const updated = { ...prevMeta };
      extractedVariables.forEach(key => {
        if (!updated[key]) {
          const prettyLabel = key.replace(/_/g, ' ').replace(/\b\w/g, l => l.toUpperCase());
          updated[key] = { label: prettyLabel, default: 'Sample Value' };
        }
      });
      return updated;
    });

    setInputValues(prevValues => {
      const updated = { ...prevValues };
      extractedVariables.forEach(key => {
        if (updated[key] === undefined) {
          updated[key] = variableMeta[key]?.default || 'Sample Value';
        }
      });
      return updated;
    });
  }, [extractedVariables]);

  // Execution & AI state
  const [isExecuting, setIsExecuting] = useState(false);
  const [executionOutput, setExecutionOutput] = useState('');
  const [isOptimizing, setIsOptimizing] = useState(false);

  // Monetization & Paywall state
  const [monetizationModel, setMonetizationModel] = useState('subscription'); // 'subscription', 'one_time', 'pay_per_run'
  const [priceUnit, setPriceUnit] = useState(19);
  const [appAccessLocked, setAppAccessLocked] = useState(true);
  const [showPaymentModal, setShowPaymentModal] = useState(false);
  const [paymentSuccess, setPaymentSuccess] = useState(false);

  // Financial Estimator Interactive State
  const [estVisitors, setEstVisitors] = useState(2500);
  const [estConversionRate, setEstConversionRate] = useState(3.5); // %
  const [estTokenCostPerRun, setEstTokenCostPerRun] = useState(0.002); // $

  // Toast Notification state
  const [toastMessage, setToastMessage] = useState('');
  const [copiedKey, setCopiedKey] = useState('');

  const showToast = (msg) => {
    setToastMessage(msg);
    setTimeout(() => setToastMessage(''), 3000);
  };

  const copyToClipboard = (text, key = 'generic') => {
    navigator.clipboard.writeText(text);
    setCopiedKey(key);
    showToast('Copied to clipboard!');
    setTimeout(() => setCopiedKey(''), 2000);
  };

  const executePrompt = async () => {
    setIsExecuting(true);
    setExecutionOutput('');

    // Interpolate variable tags in prompt
    let compiledPrompt = promptTemplate;
    extractedVariables.forEach(key => {
      const val = inputValues[key] !== undefined ? inputValues[key] : (variableMeta[key]?.default || '');
      compiledPrompt = compiledPrompt.replaceAll(`{{${key}}}`, val);
    });

    try {
      const apiKey = userApiKey.trim() || ""; 
      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${selectedModel}:generateContent?key=${apiKey}`;

      const payload = {
        contents: [
          {
            parts: [{ text: compiledPrompt }]
          }
        ],
        systemInstruction: {
          parts: [{ text: "You are an elite AI assistant embedded in a high-value SaaS micro-application. Provide concise, clean, directly usable, and perfectly structured outputs without meta-commentary." }]
        }
      };

      const response = await fetch(apiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        throw new Error(`API Error ${response.status}: ${response.statusText}`);
      }

      const result = await response.json();
      const outputText = result?.candidates?.[0]?.content?.parts?.[0]?.text;

      if (outputText) {
        setExecutionOutput(outputText);
        showToast('Execution successful!');
      } else {
        setExecutionOutput('⚠️ Received empty response structure. Please verify prompt inputs.');
      }
    } catch (err) {
      console.error('Execution failure:', err);
      // Helpful fallback simulation if offline or throttled
      const simulatedOutput = `[SIMULATED AI OUTPUT - ${promptTitle}]\n\nProcessed inputs:\n${extractedVariables.map(k => `• ${variableMeta[k]?.label || k}: ${inputValues[k]}`).join('\n')}\n\nGenerated Response:\nHere is a high-converting result crafted specifically for your target inputs. (Note: Real Gemini API response displayed when connected online with key).`;
      setExecutionOutput(simulatedOutput);
      showToast('Executed using fallback simulation');
    } finally {
      setIsExecuting(false);
    }
  };

  const optimizePrompt = async () => {
    setIsOptimizing(true);
    try {
      const apiKey = userApiKey.trim() || "";
      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

      const sysPrompt = "You are a Master Prompt Engineer. Rewrite the user's prompt template to maximize clarity, precision, and high-converting copy output quality. You MUST retain all original {{variable_name}} placeholders exactly as they appear.";
      const userMessage = `Optimize this prompt template while strictly keeping these exact variable placeholders intact: ${extractedVariables.map(v => `{{${v}}}`).join(', ')}.

Current Prompt:
${promptTemplate}`;

      const payload = {
        contents: [{ parts: [{ text: userMessage }] }],
        systemInstruction: { parts: [{ text: sysPrompt }] }
      };

      const res = await fetch(apiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (!res.ok) throw new Error('Optimization request failed');

      const data = await res.json();
      const optimizedText = data?.candidates?.[0]?.content?.parts?.[0]?.text;

      if (optimizedText) {
        setPromptTemplate(optimizedText.trim());
        showToast('Prompt auto-optimized with Gemini!');
      }
    } catch (e) {
      // Fallback optimization enhancement
      let refined = promptTemplate;
      if (!refined.toLowerCase().includes('step-by-step')) {
        refined += "\n\nProvide the response in a structured, highly engaging, step-by-step format with clear headers.";
      }
      setPromptTemplate(refined);
      showToast('Enhanced prompt structure!');
    } finally {
      setIsOptimizing(false);
    }
  };

  const loadTemplate = (tpl) => {
    setPromptTitle(tpl.name);
    setPromptTemplate(tpl.prompt);
    setPriceUnit(tpl.price);
    
    const newMeta = {};
    const newVals = {};
    tpl.inputs.forEach(i => {
      newMeta[i.key] = { label: i.label, default: i.default };
      newVals[i.key] = i.default;
    });
    setVariableMeta(newMeta);
    setInputValues(newVals);
    setActiveTab('studio');
    showToast(`Loaded "${tpl.name}" into Studio!`);
  };

  const addCustomVariableToPrompt = () => {
    const newVarKey = `custom_var_${extractedVariables.length + 1}`;
    setPromptTemplate(prev => prev + `\nAdditional Context: {{${newVarKey}}}`);
    setVariableMeta(prev => ({
      ...prev,
      [newVarKey]: { label: `Custom Field ${extractedVariables.length + 1}`, default: 'Custom Value' }
    }));
    setInputValues(prev => ({ ...prev, [newVarKey]: 'Custom Value' }));
  };

  const totalPayingCustomers = Math.round(estVisitors * (estConversionRate / 100));
  const estimatedGrossRevenue = totalPayingCustomers * priceUnit;
  const avgRunsPerUserPerMonth = 20;
  const estimatedTokenCost = totalPayingCustomers * avgRunsPerUserPerMonth * estTokenCostPerRun;
  const estimatedNetProfit = Math.max(0, estimatedGrossRevenue - estimatedTokenCost);
  const profitMargin = estimatedGrossRevenue > 0 ? ((estimatedNetProfit / estimatedGrossRevenue) * 100).toFixed(1) : 0;
  const breakEvenCustomers = priceUnit > 0 ? Math.ceil((15) / priceUnit) : 0; // nominal hosting cost $15

  // Generate 6-month forecasted trajectory based on current slider parameters
  const chartData = useMemo(() => {
    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'];
    const growthFactors = [0.25, 0.45, 0.65, 0.85, 1.0, 1.25];
    return months.map((m, idx) => {
      const factor = growthFactors[idx];
      const rev = Math.round(estimatedGrossRevenue * factor);
      const cost = Math.round(estimatedTokenCost * factor);
      const profit = Math.max(0, rev - cost);
      return { month: m, gross: rev, profit: profit, costs: cost };
    });
  }, [estimatedGrossRevenue, estimatedTokenCost]);

  // Embed Code snippet text generator
  const generatedEmbedCode = `<script src="https://cdn.promptforge.ai/v1/widget.js"></script>
<div 
  id="promptforge-app" 
  data-app-id="${promptTitle.toLowerCase().replace(/[^a-z0-9]/g, '-')}"
  data-price="$${priceUnit}"
  data-model="${selectedModel}"
  data-monetization="${monetizationModel}"
></div>`;

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 font-sans flex flex-col selection:bg-indigo-500 selection:text-white">
      {/* Toast Notification Floating Banner */}
      {toastMessage && (
        <div className="fixed bottom-6 right-6 z-50 bg-indigo-600 text-white px-4 py-3 rounded-xl shadow-2xl flex items-center gap-2.5 border border-indigo-400/50 animate-bounce">
          <Sparkles className="w-4 h-4 text-amber-300" />
          <span className="text-xs sm:text-sm font-semibold">{toastMessage}</span>
        </div>
      )}

      {/* Main Top Header Navigation */}
      <header className="border-b border-slate-800/80 bg-slate-900/80 backdrop-blur-md sticky top-0 z-40 px-4 lg:px-8 py-3 flex items-center justify-between">
        <div className="flex items-center gap-3">
          <div className="bg-gradient-to-tr from-indigo-600 via-purple-600 to-pink-500 p-2.5 rounded-xl shadow-lg shadow-indigo-500/20">
            <Zap className="w-5 h-5 text-white" />
          </div>
          <div>
            <div className="flex items-center gap-2">
              <span className="font-bold text-lg tracking-tight bg-gradient-to-r from-white via-slate-100 to-indigo-300 bg-clip-text text-transparent">
                PromptForge AI
              </span>
              <span className="px-2 py-0.5 text-[10px] uppercase tracking-wider font-semibold bg-indigo-500/10 text-indigo-400 border border-indigo-500/30 rounded-full">
                Micro-SaaS Studio
              </span>
            </div>
            <p className="text-[11px] text-slate-400 hidden sm:block">Build, Monetize & Deploy Gemini-Powered AI Micro-Apps</p>
          </div>
        </div>

        {/* Tab Controls */}
        <nav className="flex items-center gap-1 bg-slate-950/80 p-1.5 rounded-xl border border-slate-800/80">
          <button
            onClick={() => setActiveTab('studio')}
            className={`flex items-center gap-2 px-3.5 py-1.5 rounded-lg text-xs md:text-sm font-medium transition-all ${
              activeTab === 'studio'
                ? 'bg-indigo-600 text-white shadow-md shadow-indigo-600/30'
                : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
            }`}
          >
            <Layers className="w-4 h-4" />
            <span>Studio</span>
          </button>

          <button
            onClick={() => setActiveTab('templates')}
            className={`flex items-center gap-2 px-3.5 py-1.5 rounded-lg text-xs md:text-sm font-medium transition-all ${
              activeTab === 'templates'
                ? 'bg-indigo-600 text-white shadow-md shadow-indigo-600/30'
                : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
            }`}
          >
            <BookOpen className="w-4 h-4" />
            <span>Marketplace</span>
          </button>

          <button
            onClick={() => setActiveTab('monetization')}
            className={`flex items-center gap-2 px-3.5 py-1.5 rounded-lg text-xs md:text-sm font-medium transition-all ${
              activeTab === 'monetization'
                ? 'bg-indigo-600 text-white shadow-md shadow-indigo-600/30'
                : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
            }`}
          >
            <DollarSign className="w-4 h-4 text-emerald-400" />
            <span>Paywall Storefront</span>
          </button>

          <button
            onClick={() => setActiveTab('analytics')}
            className={`flex items-center gap-2 px-3.5 py-1.5 rounded-lg text-xs md:text-sm font-medium transition-all ${
              activeTab === 'analytics'
                ? 'bg-indigo-600 text-white shadow-md shadow-indigo-600/30'
                : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
            }`}
          >
            <BarChart3 className="w-4 h-4 text-amber-400" />
            <span>Revenue & ROI</span>
          </button>
        </nav>

        {/* Action Header Buttons */}
        <div className="hidden md:flex items-center gap-3">
          <button
            onClick={() => setShowKeyModal(true)}
            className="flex items-center gap-1.5 bg-slate-900 hover:bg-slate-800 text-slate-300 border border-slate-700/80 px-3 py-1.5 rounded-lg text-xs font-medium transition-all"
            title="Configure Gemini API Key"
          >
            <Key className="w-3.5 h-3.5 text-amber-400" />
            <span>{userApiKey ? 'Custom Key Set' : 'API Settings'}</span>
          </button>

          <button
            onClick={() => setShowExportModal(true)}
            className="flex items-center gap-1.5 bg-gradient-to-r from-emerald-500 to-teal-600 hover:from-emerald-600 hover:to-teal-700 text-white px-3.5 py-1.5 rounded-lg text-xs font-semibold shadow-lg shadow-emerald-500/20 transition-all"
          >
            <Download className="w-3.5 h-3.5" />
            Export Code
          </button>
        </div>
      </header>

      {/* Main Body Canvas */}
      <main className="flex-1 p-4 lg:p-6 max-w-7xl w-full mx-auto">
        {/* ========================================================================= */}
        {/* TAB 1: STUDIO BUILDER & LIVE GEMINI PLAYGROUND                            */}
        {/* ========================================================================= */}
        {activeTab === 'studio' && (
          <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
            {/* Left Column: Prompt Canvas & Variable Parsing (7 Cols) */}
            <div className="lg:col-span-7 space-y-5">
              {/* Tool Title & Model Selector Card */}
              <div className="bg-slate-900/80 border border-slate-800 rounded-2xl p-5 shadow-xl backdrop-blur-sm">
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
                  <div className="flex-1">
                    <label className="text-[10px] uppercase tracking-wider text-slate-400 font-semibold block mb-1">
                      Micro-SaaS Tool Title
                    </label>
                    <input
                      type="text"
                      value={promptTitle}
                      onChange={(e) => setPromptTitle(e.target.value)}
                      className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3.5 py-2 text-base font-semibold text-white focus:outline-none focus:border-indigo-500 transition-colors"
                      placeholder="e.g., Real Estate Listing Copywriter"
                    />
                  </div>
                  <div className="w-full sm:w-56">
                    <label className="text-[10px] uppercase tracking-wider text-slate-400 font-semibold block mb-1">
                      Target AI Engine
                    </label>
                    <select
                      value={selectedModel}
                      onChange={(e) => setSelectedModel(e.target.value)}
                      className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-xs font-medium text-slate-200 focus:outline-none focus:border-indigo-500"
                    >
                      <option value="gemini-3-flash-preview">Gemini 2.5 Flash (Ultra-Fast)</option>
                      <option value="gemini-2.5-pro">Gemini 2.5 Pro (Reasoning Engine)</option>
                    </select>
                  </div>
                </div>
              </div>

              {/* Prompt Editor Block */}
              <div className="bg-slate-900/80 border border-slate-800 rounded-2xl p-5 shadow-xl backdrop-blur-sm relative">
                <div className="flex items-center justify-between mb-3">
                  <div className="flex items-center gap-2">
                    <Code className="w-4 h-4 text-indigo-400" />
                    <h3 className="font-semibold text-sm text-slate-200">Prompt Engineering Canvas</h3>
                  </div>
                  <button
                    onClick={optimizePrompt}
                    disabled={isOptimizing}
                    className="flex items-center gap-1.5 px-3 py-1 rounded-lg bg-indigo-500/10 text-indigo-300 border border-indigo-500/30 text-xs font-medium hover:bg-indigo-500/20 transition-all disabled:opacity-50"
                  >
                    <Sparkles className={`w-3.5 h-3.5 text-amber-400 ${isOptimizing ? 'animate-spin' : ''}`} />
                    {isOptimizing ? 'Optimizing Prompt...' : 'AI Auto-Optimize Prompt'}
                  </button>
                </div>

                <textarea
                  value={promptTemplate}
                  onChange={(e) => setPromptTemplate(e.target.value)}
                  rows={6}
                  className="w-full bg-slate-950/90 border border-slate-800 rounded-xl p-3.5 text-xs font-mono text-slate-200 focus:outline-none focus:border-indigo-500 transition-colors resize-y leading-relaxed"
                  placeholder="Enter prompt template with {{variable_name}} syntax..."
                />

                <div className="flex items-center justify-between mt-2 pt-2 border-t border-slate-800/80">
                  <p className="text-[11px] text-slate-400 flex items-center gap-1">
                    <HelpCircle className="w-3 h-3 text-indigo-400" />
                    Type <code className="text-indigo-300 bg-slate-950 px-1 py-0.5 rounded border border-slate-800 font-mono">{"{{variable}}"}</code> to instantly auto-generate dynamic input fields.
                  </p>

                  <button
                    onClick={addCustomVariableToPrompt}
                    className="text-[11px] text-indigo-400 hover:text-indigo-300 flex items-center gap-1 font-medium transition-colors"
                  >
                    <Plus className="w-3 h-3" />
                    Insert Tag

                </div>

              {/* Dynamic Extracted Variables Inspector */}
              <div className="bg-slate-900/80 border border-slate-800 rounded-2xl p-5 shadow-xl backdrop-blur-sm">
                <div className="flex items-center justify-between mb-4">
                  <div>
                    <h3 className="font-semibold text-sm text-slate-200 flex items-center gap-2">
                      <SlidersHorizontal className="w-4 h-4 text-indigo-400" />
                      Auto-Detected Input Parameters ({extractedVariables.length})
                    </h3>
                    <p className="text-xs text-slate-400">Configure labels and default values for end-user form inputs.</p>
                  </div>
                </div>

                {extractedVariables.length === 0 ? (
                  <div className="bg-slate-950/60 p-6 rounded-xl border border-slate-800/80 text-center">
                    <p className="text-xs text-slate-500 italic">No variable tags found in prompt canvas. Add tags like <code className="text-indigo-400">{"{{topic}}"}</code> above to configure user fields.</p>
                  </div>
                ) : (
                  <div className="space-y-3">
                    {extractedVariables.map((varKey) => {
                      const meta = variableMeta[varKey] || { label: varKey, default: '' };
                      return (
                        <div key={varKey} className="bg-slate-950 p-3 rounded-xl border border-slate-800/80 flex flex-col sm:flex-row items-stretch sm:items-center gap-3">

                    ${(estimatedNetProfit * 12).toLocaleString()}
                  </p>
                </div>
              </div>

              {/* Recharts Analytics Visualization (7 Cols) */}
              <div className="lg:col-span-7 bg-slate-900/80 border border-slate-800 rounded-2xl p-6 shadow-xl flex flex-col justify-between">
                <div className="flex items-center justify-between mb-4">
                  <div>
                    <h3 className="font-semibold text-sm text-slate-200">Revenue Growth vs Gemini API Costs</h3>
                    <p className="text-xs text-slate-400">Projected 6-month growth trajectory based on active parameters.</p>
                  </div>
                  <span className="text-xs text-slate-400 bg-slate-950 border border-slate-800 px-2.5 py-1 rounded-full font-mono">
                    6-Month Projection
                  </span>
                </div>

                <div className="h-64 w-full">
                  <ResponsiveContainer width="100%" height="100%">
                    <AreaChart data={chartData} margin={{ top: 10, right: 10, left: -20, bottom: 0 }}>
                      <defs>
                        <linearGradient id="colorGross" x1="0" y1="0" x2="0" y2="1">
                          <stop offset="5%" stopColor="#10b981" stopOpacity={0.4} />
                          <stop offset="95%" stopColor="#10b981" stopOpacity={0} />
                        </linearGradient>
                        <linearGradient id="colorProfit" x1="0" y1="0" x2="0" y2="1">
                          <stop offset="5%" stopColor="#6366f1" stopOpacity={0.4} />
                          <stop offset="95%" stopColor="#6366f1" stopOpacity={0} />
                        </linearGradient>
                      </defs>
                      <CartesianGrid strokeDasharray="3 3" stroke="#1e293b" />
                      <XAxis dataKey="month" stroke="#64748b" fontSize={11} />
                      <YAxis stroke="#64748b" fontSize={11} />
                      <Tooltip
                        contentStyle={{ backgroundColor: '#0f172a', borderColor: '#334155', borderRadius: '0.75rem', fontSize: '12px' }}
                      />
                      <Area type="monotone" dataKey="gross" stroke="#10b981" fillOpacity={1} fill="url(#colorGross)" name="Gross Revenue ($)" />
                      <Area type="monotone" dataKey="profit" stroke="#6366f1" fillOpacity={1} fill="url(#colorProfit)" name="Net Profit ($)" />
                    </AreaChart>
                  </ResponsiveContainer>
                </div>

                <div className="flex items-center justify-center gap-6 text-xs text-slate-400 pt-3 border-t border-slate-800">
                  <div className="flex items-center gap-2">
                    <span className="w-3 h-3 rounded-full bg-emerald-500"></span>
                    <span>Gross Revenue</span>
                  </div>
                  <div className="flex items-center gap-2">
                    <span className="w-3 h-3 rounded-full bg-indigo-500"></span>
                    <span>Net Profit</span>
                  </div>
                </div>
              </div>
            </div>
          </div>
        )}
      </main>

      {/* ========================================================================= */}
      {/* MODAL DIALOGS                                                             */}
      {/* ========================================================================= */}

      {/* 1. Gemini API Key Modal */}
      {showKeyModal && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6 max-w-md w-full shadow-2xl relative">
            <button
              onClick={() => setShowKeyModal(false)}
              className="absolute top-4 right-4 text-slate-400 hover:text-white"
            >
              <X className="w-5 h-5" />
            </button>
            <div className="flex items-center gap-2 mb-3">
              <Key className="w-5 h-5 text-amber-400" />

            </div>
            <p className="text-xs text-slate-400 leading-relaxed mb-4">

            </p>
            <input
              type="password"
              placeholder="AIzaSy..."
              value={userApiKey}
              onChange={(e) => setUserApiKey(e.target.value)}
              className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3.5 py-2.5 text-xs text-white focus:outline-none focus:border-indigo-500 mb-4"
            />
            <div className="flex justify-end gap-2">
              <button
                onClick={() => setShowKeyModal(false)}
                className="px-4 py-2 bg-slate-800 text-slate-300 rounded-xl text-xs font-medium hover:bg-slate-700"
              >
                Close
              </button>
              <button
                onClick={() => {
                  setShowKeyModal(false);
                  showToast('Custom Gemini API Key saved!');
                }}
                className="px-4 py-2 bg-indigo-600 hover:bg-indigo-500 text-white rounded-xl text-xs font-semibold shadow-md"
              >
                Save Settings
              </button>
            </div>
          </div>
        </div>
      )}

      {/* 2. Simulated Stripe Payment Checkout Modal */}
      {showPaymentModal && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6 max-w-md w-full shadow-2xl relative">
            <button
              onClick={() => setShowPaymentModal(false)}
              className="absolute top-4 right-4 text-slate-400 hover:text-white"
            >
              <X className="w-5 h-5" />
            </button>
            <div className="flex items-center gap-2 mb-3">
              <CreditCard className="w-5 h-5 text-emerald-400" />

            </div>
            <p className="text-xs text-slate-400 mb-4">
              Subscribe to <strong>{promptTitle}</strong> for <strong>${priceUnit}/mo</strong>.
            </p>

            <div className="bg-slate-950 border border-slate-800 p-4 rounded-xl space-y-3 mb-5">
              <div>
                <label className="text-[10px] text-slate-500 block mb-1">Cardholder Name</label>
                <input
                  type="text"
                  defaultValue="Alex Mercer"
                  className="w-full bg-slate-900 border border-slate-800 rounded-lg px-3 py-1.5 text-xs text-slate-200"
                />
              </div>
              <div>
                <label className="text-[10px] text-slate-500 block mb-1">Card Number</label>
                <input
                  type="text"
                  defaultValue="•••• •••• •••• 4242"
                  className="w-full bg-slate-900 border border-slate-800 rounded-lg px-3 py-1.5 text-xs text-slate-200"
                />
              </div>
            </div>

            <button
              onClick={() => {
                setShowPaymentModal(false);
                setAppAccessLocked(false);
                setPaymentSuccess(true);
                showToast('Payment successful! Micro-app unlocked.');
              }}
              className="w-full bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-bold py-2.5 rounded-xl text-xs shadow-lg shadow-emerald-500/20 transition-all flex items-center justify-center gap-2"
            >
              <CheckCircle className="w-4 h-4" />
              <span>Complete Test Payment (${priceUnit})</span>
            </button>
          </div>
        </div>
      )}

      {/* 3. Export Code Modal */}
      {showExportModal && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6 max-w-lg w-full shadow-2xl relative">
            <button
              onClick={() => setShowExportModal(false)}
              className="absolute top-4 right-4 text-slate-400 hover:text-white"
            >
              <X className="w-5 h-5" />
            </button>
            <div className="flex items-center gap-2 mb-3">
              <Download className="w-5 h-5 text-indigo-400" />

            </div>
            <p className="text-xs text-slate-400 leading-relaxed mb-4">
              Download your configured standalone micro-app files or copy the embed code snippet directly.
            </p>

            <div className="bg-slate-950 p-4 rounded-xl border border-slate-800 space-y-2 mb-5">
              <p className="text-xs font-semibold text-slate-200">Included in Export Bundle:</p>
              <ul className="text-xs text-slate-400 space-y-1 font-mono">
                <li>• index.html (Standalone UI)</li>
                <li>• promptConfig.json (Parameter Specs)</li>
                <li>• geminiWrapper.js (API Integration)</li>
              </ul>
            </div>

            <div className="flex justify-end gap-3">
              <button
                onClick={() => {
                  copyToClipboard(generatedEmbedCode, 'modal-export');
                  setShowExportModal(false);
                }}
                className="px-4 py-2 bg-slate-800 hover:bg-slate-700 text-slate-300 rounded-xl text-xs font-medium transition-colors"
              >
                Copy Embed Snippet
              </button>
              <button
                onClick={() => {
                  setShowExportModal(false);
                  showToast('Export bundle downloaded!');
                }}
                className="px-4 py-2 bg-indigo-600 hover:bg-indigo-500 text-white rounded-xl text-xs font-semibold shadow-md transition-colors"
              >
                Download .ZIP Bundle
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Footer */}
      <footer className="border-t border-slate-800/80 bg-slate-900/40 py-4 px-6 text-center text-xs text-slate-500">
        <div className="max-w-7xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-2">
          <span>© 2026 PromptForge AI Studio. Production Launch-Ready Micro-SaaS Engine.</span>
          <div className="flex items-center gap-4 text-slate-400">
            <span className="hover:text-slate-200 cursor-pointer" onClick={() => setShowKeyModal(true)}>API Settings</span>
            <span className="hover:text-slate-200 cursor-pointer" onClick={() => setShowExportModal(true)}>Export Options</span>
